import java.rmi.*;
import java.rmi.server.*;

public class ServerImpl extends UnicastRemoteObject
	interface ServerIntf
{
	public double Addition(double num1,num2) throws RemoteException
	{
		return num1+num2;
	}
	public double Substraction(double num1,num2) throws RemoteException
	{
		return num1-num2;
	}
	public double Multiplication(double num1,num2) throws RemoteException
	{
		return num1*num2;
	}
	public double Division(double num1,num2) throws RemoteException
	{
		if(num2 != 0)
		{
			return num1/num2;
		}
		else
		{
			System.out.println("Cannot divide by zero");
		}
	}
}
